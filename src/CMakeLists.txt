cmake_minimum_required(VERSION 3.28)
project(biblerunner LANGUAGES CXX)

set(KF6_MIN_VERSION "5.90")

# Include the Extra-CMake-Modules project
find_package(ECM ${KF6_MIN_VERSION} REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH} ${ECM_KDE_MODULE_DIR} ${CMAKE_MODULE_PATH})

include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(FeatureSummary)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(KF6 ${KF6_MIN_VERSION} REQUIRED COMPONENTS I18n Config Runner KIO Notifications KCMUtils WidgetsAddons)

set(MODULES_FILES
    modules/query.cppm
    modules/config.cppm
)

# This takes care of building and installing the plugin
kcoreaddons_add_plugin(biblerunner
        SOURCES
        biblerunner.cpp biblerunner.h
        ${MODULES_FILES}
        INSTALL_NAMESPACE "kf6/krunner")
# Module interface units must be tracked in a CXX_MODULES file set
target_sources(biblerunner PRIVATE FILE_SET cxx_modules TYPE CXX_MODULES FILES ${MODULES_FILES})
# The plugin TU imports the module, so dependency scanning is required
set_target_properties(biblerunner PROPERTIES CXX_SCAN_FOR_MODULES ON)

# modules/books.cpp parses the JSON alias tables at compile time with C++26
# static reflection (vendored simdjson, see the file header), which clang does
# not implement yet: that TU is compiled into a standalone object by g++ and
# linked into the clang-built module target. It is not a module unit; the
# consumers include modules/books.h in their global module fragments.
#
# There is no directory reflection in C++, and #embed needs its operands as
# source-level literals, so every src/data/books/*.json is collected here:
# the glob below generates books_tables.h with one embedded table and one
# parsed document per file. CONFIGURE_DEPENDS re-globs at every build, so a
# new or removed file is picked up by the next build without further changes.
# The tables are name-sorted: when two files map the same alias to different
# books, the earlier file wins (e.g. "jn" stays the Polish JHN, not pt-br's
# JON).
file(GLOB BOOK_TABLE_GLOB RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/data/books"
     CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/data/books/*.json")
list(SORT BOOK_TABLE_GLOB)

set(BOOK_TABLES_H "${CMAKE_CURRENT_BINARY_DIR}/books_tables.h")
set(BOOK_TABLE_DEPS "")
set(BOOK_TABLES_BODY
    "#pragma once\n\n// Generated by src/CMakeLists.txt: one embedded, compile-time-parsed book\n// alias table per src/data/books/*.json (file-name order). Do not edit\n// by hand; CMake regenerates it on every build from the directory contents.\n")
list(LENGTH BOOK_TABLE_GLOB BOOK_TABLE_COUNT)
string(APPEND BOOK_TABLES_BODY "#define BOOK_TABLE_COUNT ${BOOK_TABLE_COUNT}\n\n")
set(BOOK_TABLES_DOCS "")
set(BOOK_TABLE_INDEX 0)
foreach(BOOK_TABLE_NAME IN LISTS BOOK_TABLE_GLOB)
    set(BOOK_TABLE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/data/books/${BOOK_TABLE_NAME}")
    string(APPEND BOOK_TABLES_BODY
           "constexpr char BOOK_TABLE_${BOOK_TABLE_INDEX}[]{\n"
           "#embed \"${BOOK_TABLE_PATH}\"\n"
           ", 0 };\n"
           "constexpr auto BOOK_DOC_${BOOK_TABLE_INDEX} = "
           "simdjson::compile_time::parse_json<BOOK_TABLE_${BOOK_TABLE_INDEX}>();\n\n")
    if(BOOK_TABLES_DOCS)
        string(APPEND BOOK_TABLES_DOCS ", ")
    endif()
    string(APPEND BOOK_TABLES_DOCS "BOOK_DOC_${BOOK_TABLE_INDEX}")
    list(APPEND BOOK_TABLE_DEPS "${BOOK_TABLE_PATH}")
    math(EXPR BOOK_TABLE_INDEX "${BOOK_TABLE_INDEX} + 1")
endforeach()
string(APPEND BOOK_TABLES_BODY "#define BOOK_DOCS ${BOOK_TABLES_DOCS}\n")
file(WRITE "${BOOK_TABLES_H}" "${BOOK_TABLES_BODY}")

find_program(BOOKS_GXX g++ REQUIRED)
set(BOOKS_OBJ "${CMAKE_CURRENT_BINARY_DIR}/books_gcc.o")
add_custom_command(OUTPUT "${BOOKS_OBJ}"
        COMMAND "${BOOKS_GXX}" -std=c++26 -freflection -Wno-narrowing
        "-I${CMAKE_CURRENT_SOURCE_DIR}/lib"
        "-I${CMAKE_CURRENT_BINARY_DIR}"
        -c "${CMAKE_CURRENT_SOURCE_DIR}/modules/books.cpp" -o "${BOOKS_OBJ}"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/modules/books.cpp"
                "${CMAKE_CURRENT_SOURCE_DIR}/lib/simdjson.h"
                "${BOOK_TABLES_H}"
                ${BOOK_TABLE_DEPS}
                "${BOOK_TABLES_H}"
        VERBATIM)
set_source_files_properties("${BOOKS_OBJ}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
target_sources(biblerunner PRIVATE "${BOOKS_OBJ}")
# We need to link the KRunner and other used libraries  to it
target_link_libraries(biblerunner
        KF6::I18n
        KF6::ConfigCore
        KF6::Runner
        KF6::KIOWidgets
        KF6::Notifications
)

# Configuration module, launched from the KRunner configuration page (see
# https://develop.kde.org/docs/plasma/krunner/#runner-configuration). It has no
# embedded JSON metadata on purpose: it is referenced from the runner metadata via
# X-KDE-ConfigModule (src/biblerunner.json) and exported with K_PLUGIN_CLASS.
# Building it here keeps the config source, metadata and install steps in one place.
kcoreaddons_add_plugin(kcm_biblerunner
        SOURCES
        kcm_biblerunner.cpp kcm_biblerunner.h
        INSTALL_NAMESPACE "kf6/krunner/kcms")
# The version selector reads the shipped bible version list from the install prefix
target_compile_definitions(kcm_biblerunner PRIVATE
        BIBLERUNNER_VERSIONS_FILE="${KDE_INSTALL_FULL_DATADIR}/biblerunner/versions.json")
target_link_libraries(kcm_biblerunner
        Qt6::Widgets
        KF6::I18n
        KF6::ConfigCore
        KF6::KCMUtils
        KF6::WidgetsAddons
)

# Bible version list consumed by the kcm_biblerunner version selector
install(FILES data/versions.json DESTINATION ${KDE_INSTALL_DATADIR}/biblerunner)

set(KDE_INSTALL_ICONDIR "/usr/share/pixmaps/" CACHE STRING "Default icon directory")
install(FILES bible-icon.png DESTINATION ${KDE_INSTALL_ICONDIR})
